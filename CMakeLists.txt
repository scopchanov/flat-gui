cmake_minimum_required(VERSION 3.16)

project(FlatGui VERSION 3.0.0 LANGUAGES CXX)

set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Widgets)

add_library(${PROJECT_NAME} SHARED)

add_subdirectory(src)

set_target_properties(${PROJECT_NAME} PROPERTIES
	DEBUG_POSTFIX "d"
)

target_link_libraries(${PROJECT_NAME} PRIVATE Qt6::Widgets)

target_compile_definitions(${PROJECT_NAME} PRIVATE FLATGUI_LIBRARY)

target_include_directories(${PROJECT_NAME} PUBLIC
	"$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>"
	"$<INSTALL_INTERFACE:include/${PROJECT_NAME}>"
)

install(TARGETS ${PROJECT_NAME}
	EXPORT ${PROJECT_NAME}Targets
	ARCHIVE DESTINATION ${PROJECT_VERSION}/lib
	LIBRARY DESTINATION ${PROJECT_VERSION}/lib
	RUNTIME DESTINATION ${PROJECT_VERSION}/lib
	INCLUDES DESTINATION ${PROJECT_VERSION}/include
)

install(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/src/
	DESTINATION ${PROJECT_VERSION}/include/${PROJECT_NAME}
	FILES_MATCHING PATTERN "*.h"
	PATTERN "*_p.h" EXCLUDE
)

install(EXPORT ${PROJECT_NAME}Targets
	FILE ${PROJECT_NAME}Targets.cmake
	NAMESPACE ${PROJECT_NAME}::
	DESTINATION ${PROJECT_VERSION}/lib/cmake/${PROJECT_NAME}
)

add_subdirectory(tests)
